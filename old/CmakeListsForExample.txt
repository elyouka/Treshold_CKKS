cmake_minimum_required(VERSION 3.20)

#CMake before using Google Benchmarks

project(OpenFHE_test LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_PREFIX_PATH "/mnt/d/openfhe_install" ${CMAKE_PREFIX_PATH})
find_package(OpenFHE REQUIRED)

set(CMAKE_BUILD_RPATH "/mnt/d/openfhe_install/lib")

add_executable(openfhe_test main.cpp)

target_link_options(openfhe_test PRIVATE "-Wl,--disable-new-dtags")

target_include_directories(openfhe_test PRIVATE
    ${OpenFHE_INCLUDE}
    ${OpenFHE_INCLUDE}/pke
    ${OpenFHE_INCLUDE}/core
    ${OpenFHE_INCLUDE}/binfhe
    ${OpenFHE_INCLUDE}/cereal
)

target_link_libraries(openfhe_test PRIVATE
    OPENFHEpke
    OPENFHEcore
    OPENFHEbinfhe
)
